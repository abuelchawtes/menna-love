    <script>

        const noBtn = document.getElementById('noBtn');

        let moveCount = 0;


        function handleNoAction(e) {

            if (e) e.preventDefault();

            moveCount++;

            

            // بعد 3 محاولات يتحول لـ "نعم"

            if (moveCount >= 3) {

                noBtn.innerText = "نعم 💕";

                noBtn.style.backgroundColor = "#28a745";

                noBtn.style.position = "static";

                noBtn.onclick = showLovePage;

                noBtn.removeEventListener('touchstart', handleNoAction);

                noBtn.removeEventListener('mouseover', handleNoAction);

                noBtn.removeEventListener('click', handleNoAction);

                return;

            }


            // أبعاد الشاشة الآمنة للموبايل

            const btnWidth = noBtn.offsetWidth || 100;

            const btnHeight = noBtn.offsetHeight || 50;

            

            const maxX = window.innerWidth - btnWidth - 20;

            const maxY = window.innerHeight - btnHeight - 20;


            const randomX = Math.max(20, Math.floor(Math.random() * maxX));

            const randomY = Math.max(20, Math.floor(Math.random() * maxY));


            noBtn.style.position = "fixed";

            noBtn.style.left = randomX + "px";

            noBtn.style.top = randomY + "px";

        }


        // إضافات خاصة باللمس للموبايل لمنع أي تأخير أو مشاكل

        noBtn.addEventListener('touchstart', handleNoAction, { passive: false });

        noBtn.addEventListener('mouseover', handleNoAction);

        noBtn.addEventListener('click', handleNoAction);


        function showLovePage() {

            document.getElementById('page1').style.display = 'none';

            document.getElementById('page2').style.display = 'block';

            createHearts();

        }


        function createHearts() {

            for (let i = 0; i < 30; i++) {

                setTimeout(() => {

                    const heart = document.createElement('div');

                    heart.classList.add('bg-heart');

                    heart.innerHTML = '❤️';

                    heart.style.left = Math.random() * 100 + 'vw';

                    heart.style.animationDuration = (Math.random() * 2 + 3) + 's';

                    heart.style.fontSize = (Math.random() * 20 + 20) + 'px';

                    document.body.appendChild(heart);

                }, i * 150);

            }

        }
